service cloud.firestore {
  match /databases/{database}/documents {
  
    match /products/{productId}{
    
      allow read;
      allow write: if request.auth.uid == request.resource.data.uid;
      
      match /comments/{commentId} {
      	allow read;
        allow write: if request.auth.uid == request.resource.data.uid;
      }
      
    }
    
    match /users/{userId}{
      allow read;
      allow write: if request.auth.uid == request.resource.data.uid;
    }
    
    match /cars/{carId}{
      allow read;
      allow write: if request.auth.uid == request.resource.data.uid;
      match /comments/{commentId} {
      	allow read;
        allow write: if request.auth.uid == request.resource.data.uid;
      }
    }
    
    match /cover/{coverId} {
    	allow read;
    }
    
  }
}
