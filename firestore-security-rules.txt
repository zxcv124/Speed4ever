service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function ownsIncomingDocument() {
      return signedIn() && request.resource.data.uid == request.auth.uid;
    }

    function ownsExistingDocument() {
      return signedIn() && resource.data.uid == request.auth.uid;
    }

    function preservesOwner() {
      return request.resource.data.uid == resource.data.uid;
    }

    function ownsUserPath(userId) {
      return signedIn()
        && get(/databases/$(database)/documents/users/$(userId)).data.uid == request.auth.uid;
    }

    match /products/{productId} {
      allow read;
      allow create: if ownsIncomingDocument();
      allow update, delete: if ownsExistingDocument() && preservesOwner();

      match /bids/{bidId} {
        allow read;
        allow write: if false;
      }

      match /comments/{commentId} {
        allow read;
        allow create: if ownsIncomingDocument();
        allow update, delete: if ownsExistingDocument() && preservesOwner();
      }
    }

    match /users/{userId} {
      allow read;
      allow create: if ownsIncomingDocument();
      allow update, delete: if ownsExistingDocument() && preservesOwner();

      match /notifications/{notificationId} {
        allow read, update, delete: if ownsUserPath(userId);
        allow create: if false;
      }
    }

    match /cars/{carId} {
      allow read;
      allow create: if ownsIncomingDocument();
      allow update, delete: if ownsExistingDocument() && preservesOwner();

      match /comments/{commentId} {
        allow read;
        allow create: if ownsIncomingDocument();
        allow update, delete: if ownsExistingDocument() && preservesOwner();
      }
    }

    match /cover/{coverId} {
      allow read;
      allow write: if false;
    }
  }
}
